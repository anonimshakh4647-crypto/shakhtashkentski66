const SHEET_NAME = 'Natijalar';

function getSheet_() {
  const ss = SpreadsheetApp.getActiveSpreadsheet();
  let sh = ss.getSheetByName(SHEET_NAME);
  if (!sh) sh = ss.insertSheet(SHEET_NAME);

  if (sh.getLastRow() === 0) {
    sh.appendRow(['Name','Email','Picture','Score','Total','Percent','Date']);
  }
  return sh;
}

function doGet(e) {
  const p = e.parameter || {};
  const sh = getSheet_();
  const values = sh.getDataRange().getValues();
  const rows = [];

  for (let i = 1; i < values.length; i++) {
    const name = String(values[i][0] || '').trim();
    if (!name) continue;

    const low = name.toLowerCase();
    if (low === 'name' || low === 'foydalanuvchi') continue;

    rows.push({
      name: name,
      picture: String(values[i][2] || ''),
      score: Number(values[i][3] || 0),
      total: Number(values[i][4] || 300),
      percent: Number(values[i][5] || 0)
    });
  }

  // Har bir Google akkaunt uchun bitta, eng so'nggi yozuv.
  // Email ustunini reytingda ko'rsatmaymiz.
  const best = {};
  for (let i = 1; i < values.length; i++) {
    const name = String(values[i][0] || '').trim();
    const email = String(values[i][1] || '').trim().toLowerCase();
    if (!name) continue;

    const low = name.toLowerCase();
    if (low === 'name' || low === 'foydalanuvchi') continue;

    const key = email || low;
    best[key] = {
      name: name,
      picture: String(values[i][2] || ''),
      score: Number(values[i][3] || 0),
      total: Number(values[i][4] || 300),
      percent: Number(values[i][5] || 0)
    };
  }

  const out = Object.values(best).sort(
    (a, b) => b.percent - a.percent || b.score - a.score || a.name.localeCompare(b.name)
  );

  const data = JSON.stringify({rows: out.slice(0, 100)});
  const cb = p.callback;

  if (cb) {
    return ContentService
      .createTextOutput(cb + '(' + data + ')')
      .setMimeType(ContentService.MimeType.JAVASCRIPT);
  }

  return ContentService
    .createTextOutput(data)
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  try {
    const body = JSON.parse(e.postData.contents || '{}');

    if (body.action !== 'save') {
      return ContentService
        .createTextOutput(JSON.stringify({ok:false}))
        .setMimeType(ContentService.MimeType.JSON);
    }

    const sh = getSheet_();

    let name = String(body.name || '').trim();
    if (!name || name.toLowerCase() === 'name') name = 'Foydalanuvchi';

    const email = String(body.email || '').trim();
    const picture = String(body.picture || '').trim();

    const total = Math.max(1, Number(body.total) || 300);
    const score = Math.max(0, Math.min(Number(body.score) || 0, total));
    const percent = Math.round(score / total * 100);

    const values = sh.getDataRange().getValues();

    // Shu akkaunt oldin yozilgan bo'lsa, yangi natija bilan shu qatorni yangilaymiz.
    // Shuning uchun har bir savoldan keyin bitta ishtirokchi uchun yangi progress saqlanadi.
    const emailKey = email.toLowerCase();
    let foundRow = -1;

    if (emailKey) {
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][1] || '').trim().toLowerCase() === emailKey) {
          foundRow = i + 1;
          break;
        }
      }
    }

    if (foundRow === -1 && name) {
      for (let i = 1; i < values.length; i++) {
        if (String(values[i][0] || '').trim().toLowerCase() === name.toLowerCase()) {
          foundRow = i + 1;
          break;
        }
      }
    }

    const row = [name, email, picture, score, total, percent, new Date()];

    if (foundRow > 1) {
      sh.getRange(foundRow, 1, 1, row.length).setValues([row]);
    } else {
      sh.appendRow(row);
    }

    return ContentService
      .createTextOutput(JSON.stringify({ok:true}))
      .setMimeType(ContentService.MimeType.JSON);

  } catch (err) {
    return ContentService
      .createTextOutput(JSON.stringify({ok:false, error:String(err)}))
      .setMimeType(ContentService.MimeType.JSON);
  }
}
